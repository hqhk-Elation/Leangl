function(leangl_add_test_variant target flush_async)
    add_library(${target} STATIC ${LEANGL_SOURCES})
    target_link_libraries(${target} PUBLIC leanGL_headers)
    target_compile_definitions(${target} PUBLIC LEANGL_FLUSH_ASYNC=${flush_async})
    leanGL_apply_build_settings(${target})
endfunction()
